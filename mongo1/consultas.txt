db.pedidos.aggregate([
  { $unwind: $lineas },
  { $match: { "lineas.cantidad": { $gt:100 } } },
  { $project: {
    _id: 0,
    idPedido: 1,
    idCliente: 1,
    fecha: 1,
    estado: 1,
    linea: "$lineas" } }
])

//Ver los pedidos cuya suma de las cantidades de las líneas cumplen >100
db.pedidos.aggregate([
  {
    $project: {
      _id: 0,
      idPedido: 1,
      idCliente: 1,
      fecha: 1,
      estado: 1,
      cantidadTotal: { $sum: "$lineas.cantidad"}
    }
  },
  {
    $match: {
      cantidadTotal: { $gt: 100}
    }
  }
])
